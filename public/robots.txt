# Gallery photos are served through the /api/photo proxy (they live in Google
# Drive, not the repo). Every one of the 227 prerendered /photo/* pages points
# its og:image and schema.org contentUrl at /api/photo?id=... — so a blanket
# "Disallow: /api/" told Google it may read those pages but must not fetch the
# photograph they exist to show. On a photography site that is the whole point
# of the page. This Allow must stay ABOVE the /api/ Disallow and be more
# specific than it; both Google and Bing resolve conflicts by longest match.
Allow: /api/photo

Disallow: /private/
Disallow: /admin
Disallow: /admin/
Disallow: /portal
Disallow: /portal/
Disallow: /api/
